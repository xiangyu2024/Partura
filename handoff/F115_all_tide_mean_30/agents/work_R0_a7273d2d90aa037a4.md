# /root/autodl-tmp/AlphaCube/src/alphacube/operators/dt_in.py:966
@op("dt", "in", arity=1, requires_window=True)
def in_local_mean(x: torch.Tensor, d: int) -> torch.Tensor:
    """`result[i, j] = nanmean(x[i, j-d+1 : j+d])` — a centered (non-causal)
    local window of radius `d-1`, unlike the causal `in_mean_rolling`."""
    if d <= 0:
        raise ValueError("d must be a positive integer")
    N, T = x.shape
    if d == 1:
        return x.clone()
    window_size = 2 * d - 1
    offsets = torch.arange(-d + 1, d, device=x.device)
    col_indices = torch.arange(T, device=x.device).unsqueeze(1) + offsets
    valid = (col_indices >= 0) & (col_indices < T)
    clamped = torch.clamp(col_indices, 0, T - 1)
    expanded = x.unsqueeze(1).expand(N, T, T)
    gathered = torch.gather(expanded, dim=2, index=clamped.unsqueeze(0).expand(N, T, window_size))
    window_vals = torch.where(valid.unsqueeze(0).expand_as(gathered), gathered, torch.full_like(gathered, float("nan")))
    return torch.nanmean(window_vals, dim=2)

# --- rolling_reduce 调用的: common.rolling_windows
def rolling_windows(x: torch.Tensor, d: int, axis: int = 0) -> torch.Tensor:
    """Causal rolling windows of length `d` along `axis`, left-padded with
    NaN so every original position (including the first `d-1`) gets a
    window. Output shape = `x.shape` with a new trailing window-dim of
    size `d` appended (the `axis` dim's length is unchanged).
    """
    d = int(d)
    pad_shape = list(x.shape)
    pad_shape[axis] = d - 1
    pad = torch.full(pad_shape, float("nan"), dtype=x.dtype, device=x.device)
    x_padded = torch.cat([pad, x], dim=axis)
    return x_padded.unfold(axis, d, 1)

  分钟成交额与分钟最低价的协方差。

`day_durbin_watson` 的形式值得写成公式,因为整条 alpha 的一半解释都在这里:

```
DW(x) = Σ_{m=2..M} (x_m − x_{m−1})²  /  ( Σ_{m=1..M} x_m² + 1e−8 )
```

分母是 **Σx²,不是方差**(探针:平移不变 = False,齐次度 = 0)。把 x = Open 写成 P(1+u_m),

```
DW(Open) ≈ (1/M) Σ r_m²  =  日内分钟收益的已实现方差
```

也就是说,这个名字叫"Durbin-Watson 自相关检验统计量"的算子,作用在**价格水平序列**上时
根本不是在检验自相关,而是在算**日内波动率的平方**。这不是猜测:把它换成 `day_var(Ret)`

   用的是**全天**的统计量(包含触发分钟之后的数据)。这在日内是「事后」阈值:一分钟算不算耀眼,要等收盘才知道。
   对一个收盘后才使用的日频因子没有泄漏问题,但它**不是**一个盘中可以实时触发的事件。
   ΔVolume 的均值 = (最后一分钟量 − 第一分钟量)/235 ≈ 0,所以阈值实际上就是「ΔVolume > 1σ」。
4. **`in_slice(x, 2, 237)` 两端都含**(`x[:, 2:238]`),241 格里取 236 分钟:去掉开盘头两格和收盘集合竞价的三格。
5. **`cs_demean` 减的是当天有值股票的简单均值**(`cs_mean` 用 `nanmean`)。`J` 右偏,均值在中位数右边
   (具体在第几分位见 `07x_look.md`),所以「离均值最近」不是「排在中间」。
6. `day_mean` 对 NaN 分钟用 `nanmean`:`J` = 当天所有触发点的 5 分钟 std 的**等权**均值,和触发了多少次无关。
